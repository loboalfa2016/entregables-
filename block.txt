# bucle infinito hasta que se reciba un nombre de producto válido
while True:
    # pide al usuario el nombre del producto y asegúrate de que sea una cadena
    producto = str(input("Escribe el nombre del producto que deseas comprar: "))
    # elimina espacios y verifica que todos los caracteres restantes sean letras
    if producto.replace(" ", "").isalpha():
        # entrada válida; salir del bucle de validación
        break
    else:
        # retroalimentación si el nombre contiene caracteres no letras
        print("Por favor, ingresa un nombre de producto válido (solo letras). Intenta de nuevo.")

# bucle hasta que se ingrese un precio positivo válido
while True:
    try:
        # lee la entrada y la convierte a flotante; lanza ValueError si es inválida
        precio = float(input("Ingresa el precio del producto: "))
        if precio > 0:
            # precio es positivo, aceptarlo
            break
        else:
            # precio cero o negativo no está permitido
            print("Error: El precio debe ser mayor a 0.")
    except:
        # maneja la entrada no numérica
        print("Error: Ingresa un número válido para el precio.")

# bucle hasta que se ingrese una cantidad entera positiva válida
while True:
    try:
        # convierte la entrada a entero; lanza ValueError si no es un entero
        cantidad = int(input("Ingresa la cantidad del producto a comprar: "))
        if cantidad > 0:
            # cantidad válida proporcionada
            break
        else:
            # cantidades cero o negativas no están permitidas
            print("Error: La cantidad debe ser mayor a 0.")
    except:
        # captura la entrada no entera
        print("Error: La cantidad debe ser un número entero.")

# calcula el costo total multiplicando el precio por unidad por la cantidad
costo_total = precio * cantidad

# muestra el costo total formateado con dos decimales
print(f"El costo total de {cantidad} {producto}(s) es: ${costo_total:.2f}")
